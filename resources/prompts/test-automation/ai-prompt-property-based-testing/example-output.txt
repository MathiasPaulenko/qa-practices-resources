from hypothesis import given, settings, strategies as st
from decimal import Decimal

TIERS = ["bronze", "silver", "gold", "platinum"]
prices = st.decimals(min_value=0, max_value=1000000, places=4)
tiers = st.sampled_from(TIERS)

@given(prices, tiers)
@settings(max_examples=500)
def test_discount_never_exceeds_price(price, tier):
    assert calculate_discount(price, tier) <= price

@given(prices, tiers)
def test_discount_non_negative(price, tier):
    assert calculate_discount(price, tier) >= 0

@given(prices)
def test_tier_ordering_monotone(price):
    d = {t: calculate_discount(price, t) for t in TIERS}
    assert d["platinum"] >= d["gold"] >= d["silver"] >= d["bronze"]

@given(prices, tiers)
def test_deterministic(price, tier):
    assert calculate_discount(price, tier) == calculate_discount(price, tier)

@given(tiers)
def test_zero_price(tier):
    assert calculate_discount(0, tier) == 0
