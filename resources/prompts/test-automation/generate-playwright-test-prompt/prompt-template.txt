## Required Test Structure

### 1. Page Object Model (POM)
Create a dedicated page object class for each major page/flow:
```typescript
// pages/LoginPage.ts
import { Page, Locator, expect } from '@playwright/test';

export class LoginPage {
  readonly usernameInput: Locator;
  readonly passwordInput: Locator;
  readonly submitButton: Locator;
  readonly errorMessage: Locator;

  constructor(private page: Page) {
    this.usernameInput = page.locator('[data-testid="username-input"]');
    this.passwordInput = page.locator('[data-testid="password-input"]');
    this.submitButton = page.locator('[data-testid="login-button"]');
    this.errorMessage = page.locator('[data-testid="error-message"]');
  }

  async login(username: string, password: string): Promise<void> {
    await this.usernameInput.fill(username);
    await this.passwordInput.fill(password);
    await this.submitButton.click();
  }

  async expectError(message: string): Promise<void> {
    await expect(this.errorMessage).toHaveText(message);
  }
}
```

### 4. Assertions
Use Playwright's built-in, auto-retrying assertions:
```typescript
// Element state
await expect(page.locator('[data-testid="success-message"]')).toBeVisible();
await expect(page.locator('[data-testid="submit-button"]')).toBeEnabled();

// Text content
await expect(page.locator('h1')).toHaveText('Order Confirmed');
await expect(page.locator('.order-id')).toContainText('ORD-');

// URL validation
await expect(page).toHaveURL(/\/checkout\/confirmation/);

// Form validation
await expect(page.locator('[data-testid="email-error"]')).toHaveText('Invalid email format');
```
